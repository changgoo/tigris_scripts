# NOTE (2026-10-04): variants with -fp-model=fast=2 or -ffast-math without
# -fno-finite-math-only remove the solver's NaN checks; they are timing-only and must not
# become FLAGS_* in env.sh (build_tigress.sh refuses them outside bench/flag_sweep.sh).
# Compiler-flag sweep for stellarai-amd (bench/flag_sweep.sh stellarai-amd).
# One variant per line:  <toolchain> <variant> <complete compiler flags>
# The flags REPLACE configure.py's preset flags entirely (build_tigress.sh --cxxflags).
# All variants use Intel MPI so that differences come from the compiler alone.
# Winners per compiler go to FLAGS_GCC / FLAGS_AOCC / FLAGS_ICPX in env.sh.

# GCC 14
gcc-impi  preset        -O3 -std=c++11 -march=znver5 -ffast-math -fopenmp-simd -flto=auto -fwhole-program -fprefetch-loop-arrays
gcc-impi  O3            -O3 -std=c++11 -march=znver5
gcc-impi  safe-lto      -O3 -std=c++11 -march=znver5 -fno-math-errno -fno-trapping-math -fopenmp-simd -flto=auto
gcc-impi  fast          -O3 -std=c++11 -march=znver5 -ffast-math -fopenmp-simd
gcc-impi  fast-lto      -O3 -std=c++11 -march=znver5 -ffast-math -fopenmp-simd -flto=auto
gcc-impi  fast-lto-wp   -O3 -std=c++11 -march=znver5 -ffast-math -fopenmp-simd -flto=auto -fwhole-program
gcc-impi  preset-v512   -O3 -std=c++11 -march=znver5 -ffast-math -fopenmp-simd -flto=auto -fwhole-program -fprefetch-loop-arrays -mprefer-vector-width=512
gcc-impi  preset-unroll -O3 -std=c++11 -march=znver5 -ffast-math -fopenmp-simd -flto=auto -fwhole-program -fprefetch-loop-arrays -funroll-loops
gcc-impi  preset-O2     -O2 -std=c++11 -march=znver5 -ffast-math -fopenmp-simd -flto=auto -fwhole-program -fprefetch-loop-arrays
