# Compiler-flag sweep for Tiger (tiger3) Sapphire Rapids nodes (bench/flag_sweep.sh tiger).
# One variant per line:  <toolchain> <variant> <complete compiler flags>
# The flags REPLACE configure.py's preset flags entirely (build_tigress.sh --cxxflags).
# Every variant of a compiler uses the same MPI (the matrix winner for that compiler), so
# differences come from the flags alone. Winners go to FLAGS_GCC / FLAGS_ICPX in env.sh.
# Sapphire Rapids has two AVX-512 FMA units per core, so 512-bit vectors are worth trying.

# GCC 11.5 (system; the only GCC on tiger). Needs the Accretion::rctrl patch in the lanes.
gcc-impi  preset        -O3 -std=c++11 -march=sapphirerapids -ffast-math -fopenmp-simd -flto=auto -fwhole-program -fprefetch-loop-arrays
gcc-impi  O3            -O3 -std=c++11 -march=sapphirerapids
gcc-impi  safe-lto      -O3 -std=c++11 -march=sapphirerapids -fno-math-errno -fno-trapping-math -fopenmp-simd -flto=auto
gcc-impi  fast          -O3 -std=c++11 -march=sapphirerapids -ffast-math -fopenmp-simd
gcc-impi  fast-lto      -O3 -std=c++11 -march=sapphirerapids -ffast-math -fopenmp-simd -flto=auto
gcc-impi  fast-lto-wp   -O3 -std=c++11 -march=sapphirerapids -ffast-math -fopenmp-simd -flto=auto -fwhole-program
gcc-impi  preset-v512   -O3 -std=c++11 -march=sapphirerapids -ffast-math -fopenmp-simd -flto=auto -fwhole-program -fprefetch-loop-arrays -mprefer-vector-width=512
gcc-impi  preset-unroll -O3 -std=c++11 -march=sapphirerapids -ffast-math -fopenmp-simd -flto=auto -fwhole-program -fprefetch-loop-arrays -funroll-loops
gcc-impi  preset-O2     -O2 -std=c++11 -march=sapphirerapids -ffast-math -fopenmp-simd -flto=auto -fwhole-program -fprefetch-loop-arrays

# oneAPI 2024.2 icpx (default fp-model=fast); "preset" is configure.py's icpx preset with
# -xhost spelled out for Sapphire Rapids (what the legacy tiger build produced).
# Without -ipo the link fails (omp declare simd variants; see stellar/flag_variants.txt).
icpx-impi preset        -O3 -std=c++11 -ipo -xSAPPHIRERAPIDS -qopenmp-simd -Wno-tautological-constant-compare -Wno-array-bounds
icpx-impi march         -O3 -std=c++11 -ipo -march=sapphirerapids -qopenmp-simd -Wno-tautological-constant-compare -Wno-array-bounds
icpx-impi fast2         -O3 -std=c++11 -ipo -xSAPPHIRERAPIDS -qopenmp-simd -fp-model=fast=2 -Wno-tautological-constant-compare -Wno-array-bounds
icpx-impi precise       -O3 -std=c++11 -ipo -xSAPPHIRERAPIDS -qopenmp-simd -fp-model=precise -Wno-tautological-constant-compare -Wno-array-bounds
icpx-impi preset-v512   -O3 -std=c++11 -ipo -xSAPPHIRERAPIDS -qopenmp-simd -qopt-zmm-usage=high -Wno-tautological-constant-compare -Wno-array-bounds
icpx-impi fast2-v512    -O3 -std=c++11 -ipo -xSAPPHIRERAPIDS -qopenmp-simd -fp-model=fast=2 -qopt-zmm-usage=high -Wno-tautological-constant-compare -Wno-array-bounds

# oneAPI 2026.0 icpx (+ Intel MPI 2021.18): the newer compiler with the same flag sets
icpx2026-impi preset    -O3 -std=c++11 -ipo -xSAPPHIRERAPIDS -qopenmp-simd -Wno-tautological-constant-compare -Wno-array-bounds
icpx2026-impi fast2     -O3 -std=c++11 -ipo -xSAPPHIRERAPIDS -qopenmp-simd -fp-model=fast=2 -Wno-tautological-constant-compare -Wno-array-bounds
